            part = bound(ref, protect=protect)

            chunk = part['value']
            assert isinstance(chunk, list if kind == 'list' else dict)

                assert not set(result).intersection(chunk)
                result.update(chunk)
            parts.append(dict(original_field=field, **ref_for(p), json_selector='value'))
        assert len(result) == value['original_entries']
        fields[field] = dict(container_type=kind, entries=len(result), parts=len(value['parts']))
        return expand(result, field, parts, fields, protect)

        return {k: expand(v, k, parts, fields, protect) for k, v in value.items()}
    if isinstance(value, list):
        return [expand(v, field, parts, fields, protect) for v in value]
    return value

def raw_root_values(raw):
    text, decoder, result, i = raw.decode(), json.JSONDecoder(), {}, 1
    assert text.startswith('{')
    while True:
        while text[i].isspace() or text[i] == ',':
            i += 1
        if text[i] == '}':
            return result
        key, i = decoder.raw_decode(text, i)
        assert key not in result
        while text[i].isspace():
            i += 1
        assert text[i] == ':'
        i += 1
        while text[i].isspace():
            i += 1
        start = i
        _, i = decoder.raw_decode(text, i)
        result[key] = text[start:i]

def fragments(text):
    decoder, result, i = json.JSONDecoder(), [], 1
    assert text.startswith('[')
    while True:
        while text[i].isspace() or text[i] == ',':
            i += 1
        if text[i] == ']':
            return result
        start = i
        _, i = decoder.raw_decode(text, i)
        result.append(text[start:i])
