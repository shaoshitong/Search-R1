def pin_nodes(value):
    if isinstance(value, list):
        for v in value:
            yield from pin_nodes(v)
    elif isinstance(value, dict):
        if digest_ok(value.get('sha256')) and type(value.get('bytes')) is int and (value.get('path') or value.get('file')):
            yield value
        else:
            for key, v in value.items():
                if isinstance(v, dict) and digest_ok(v.get('sha256')) and type(v.get('bytes')) is int and Path(key).suffix in ('.json', '.csv', '.md', '.py', '.txt', '.sh', '.yaml', '.yml', '.patch') and not (v.get('path') or v.get('file')):
                    yield dict(path=key, **v)
                else:
                    yield from pin_nodes(v)

def expand(value, field, parts, fields, protect=False):
    if isinstance(value, dict) and value.get('storage') == 'complete_original_field_parts':
        kind = value['container_type']
        assert kind in ('list', 'mapping') and value['parts']
        result, seen = ([] if kind == 'list' else {}), set()
        for ref in value['parts']:
            assert ref.get('json_selector') == 'value'
            p = locate(ref.get('path', ref.get('file')))
            assert p not in seen, 'duplicate CUM part'
            seen.add(p)
            part = bound(ref, protect=protect)
            assert part['schema'] == 'cumulative_scalar_field_part_v1' and part['original_field'] == field
            chunk = part['value']
            assert isinstance(chunk, list if kind == 'list' else dict)
            if kind == 'list':
                result.extend(chunk)
            else:
                assert not set(result).intersection(chunk)
                result.update(chunk)
            parts.append(dict(original_field=field, **ref_for(p), json_selector='value'))
        assert len(result) == value['original_entries']
        fields[field] = dict(container_type=kind, entries=len(result), parts=len(value['parts']))
        return expand(result, field, parts, fields, protect)
    if isinstance(value, dict):
        return {k: expand(v, k, parts, fields, protect) for k, v in value.items()}
    if isinstance(value, list):
        return [expand(v, field, parts, fields, protect) for v in value]
    return value

def raw_root_values(raw):
    text, decoder, result, i = raw.decode(), json.JSONDecoder(), {}, 1
    assert text.startswith('{')
    while True:
        while text[i].isspace() or text[i] == ',':
            i += 1
        if text[i] == '}':
            return result
        key, i = decoder.raw_decode(text, i)
        assert key not in result
        while text[i].isspace():
            i += 1
        assert text[i] == ':'
        i += 1
        while text[i].isspace():
            i += 1
        start = i
        _, i = decoder.raw_decode(text, i)
        result[key] = text[start:i]

def fragments(text):
    decoder, result, i = json.JSONDecoder(), [], 1
    assert text.startswith('[')
    while True:
        while text[i].isspace() or text[i] == ',':
            i += 1
        if text[i] == ']':
            return result
        start = i
        _, i = decoder.raw_decode(text, i)
        result.append(text[start:i])

def atomic(p, raw):
    q = p.with_name(p.name + '.200-closure.tmp')
    assert not q.exists(), 'stale temporary file: ' + str(q)
    with q.open('xb') as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(q, p)
